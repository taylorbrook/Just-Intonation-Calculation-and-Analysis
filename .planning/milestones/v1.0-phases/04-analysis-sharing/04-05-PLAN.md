       function rebuild(): void {
         try {
           const generator = new Interval(`${genN}/${genD}`);
           const period = new Interval(`${perN}/${perD}`);
           let effectiveSize = size;
           if (snap) {
             const snapped = nearestMosSize(generator, period, size);
             if (snapped !== size) {
               status.textContent = `Snapped to MOS size ${snapped} (target ${size})`;
             } else {
               status.textContent = "";
             }
             effectiveSize = snapped;
           } else {
             status.textContent = "";
           }
           const scale = buildMos(generator, period, effectiveSize);
           // Replace tableHost contents
           tableHost.replaceChildren(scaleTable(scale, opts.baseHz ?? 440, { precision: opts.precision ?? 1 }));
           // Replace playHost contents
           playHost.replaceChildren(playScale(scale, synth, { baseHz: opts.baseHz ?? 440 }));

           const msg = err instanceof Error ? err.message : String(err);

       }
