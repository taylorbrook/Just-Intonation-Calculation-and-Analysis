```typescript
// For an N-EDO row, render the user's scale at the EDO's nearest steps and
// arpeggiate via synth.playArpeggio. Reuses synth.playArpeggio unchanged —
// just different Hz values.
const rendered = scale.intervals.map(iv => {
  const nearestStep = Math.round(iv.cents / 1200 * N);
  return baseHz * Math.pow(2, nearestStep / N);
});
synth.playArpeggio(rendered, 0.45);  // 0.45 = playScale's default stepSec
```
