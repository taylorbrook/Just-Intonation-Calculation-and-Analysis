.scl format reference (RESEARCH.md §.scl Format Reference, lines 357–430):

```
! comment line (any number of these, anywhere)
<description, exactly one line — first non-comment line; can be empty>
<pitch count, second non-comment line — integer; EXCLUDES the implicit 1/1>
```

|-------|------|---------|
| `5/4` | ratio | 5/4 |
| `5` | ratio | 5/1 (bare integer) |
| `408.0` | cents | 408.0¢ |
| `408.` | cents | 408.0¢ (trailing dot) |
| `.5` | cents | 0.5¢ (leading dot) |
| `-5.0` | cents | -5.0¢ (cents may be negative) |
| `100.0 cents` | cents | 100.0¢ — anything after the value is ignored |
| ` 5/4   E\\ ` | ratio | 5/4 — surrounding whitespace and trailing names ignored |
| `2/3/4` | INVALID | multi-slash → error |

    Step 2 — Implement src/lib/scala.ts:

    ```ts
    /**
     * Scala .scl parser + serializer. Handles ratios (5/4), cents (408.0 — `.` triggers
     * cents per D-19), and the project-specific monzo bra-ket notation [-2 0 1> (D-15).
     *
     * Parser semantics per Huygens-Fokker spec + project decisions:
     * - `!` lines are comments (anywhere); ignored
     * - First non-comment line is the description (may be empty)
     * - Second non-comment line is the pitch count (excluding the implicit 1/1)
     * - Subsequent non-comment lines are pitches; auto-prepend 1/1 per D-13
     * - Last pitch line IS the period (D-14)
     * - Negative ratios (-5/4) and multi-slash ratios (2/3/4) MUST error (Pitfall #6)
     * - BOM is stripped; CRLF is normalized to LF
     * - Trailing text after a token (e.g. "5/4   E\\\\") is ignored (F09)
     *
     * R-01: All Interval constructions go through Interval (which uses fraction.js BigInt).
     * Cents-source pitches are LOSSY — the parser converts cents to a ratio via centsToValue,
     * acknowledging that the resulting Fraction is a float-derived approximation. For
     * round-trip purposes, writeScl re-emits cents (NOT ratio) for cents-source intervals
     * via the `source` metadata recorded at parse time.
     */

    import { Interval } from "./interval.js";
    import { Scale } from "./scale.js";
    import { centsToValue } from "xen-dev-utils";   // R-01 NOTE: only the helper, NOT Fraction

    /**
     * Parse the body (pitch lines only — no description / count header) of a Scala file
     * or text-input. Auto-prepends 1/1 (D-13). Comment lines (`!`-prefixed) are skipped.
     * Empty lines are skipped.
     */
    export function parseScala(body: string): Interval[] {
      const intervals: Interval[] = [new Interval("1/1")];
      const lines = normalizeLines(body);
      for (const raw of lines) {
        const line = stripComment(raw).trim();
        if (line === "") continue;
        intervals.push(parsePitchToken(line));
      }
      return intervals;
    }

    export interface ParsedScl {
      description: string;
      intervals: Interval[];
    }

    /**
     * Parse a full .scl file. Auto-prepends 1/1 (D-13). Last interval is the period (D-14).
     */
    export function parseScl(file: string): ParsedScl {
      const lines = normalizeLines(file);
      const nonComment: string[] = [];
      for (const raw of lines) {
        const stripped = raw.replace(/^﻿/, ""); // BOM (only effective on first line)
        if (stripped.startsWith("!")) continue;
        nonComment.push(stripped);
      }
      if (nonComment.length < 2) {
        throw new Error("parseScl: file is missing description and/or pitch-count header");
      }
      const description = nonComment[0]!.trim();
      const countLine = nonComment[1]!.trim();
      const expectedCount = parseInt(countLine, 10);
      if (!Number.isFinite(expectedCount) || expectedCount < 0) {
        throw new Error(`parseScl: invalid pitch count "${countLine}"`);
      }
      const pitchLines = nonComment.slice(2).map((l) => stripComment(l).trim()).filter((l) => l !== "");
      if (pitchLines.length !== expectedCount) {
        throw new Error(
          `parseScl: pitch count mismatch — header says ${expectedCount}, found ${pitchLines.length} pitch lines`,
        );
      }
      const intervals: Interval[] = [new Interval("1/1")];
      for (const line of pitchLines) intervals.push(parsePitchToken(line));
      return { description, intervals };
    }

    /**
     * Serialize a Scale to .scl format. Does NOT emit the unison line (D-13).
     * Last interval is treated as the period (D-14).
     */
    export function writeScl(scale: Scale, description?: string): string {
      // Strip leading 1/1 if present (D-13 contract — the parser auto-prepends, the serializer strips).
      const intervals = scale.intervals.slice();
      if (intervals.length > 0 && intervals[0]!.equals(new Interval("1/1"))) {
        intervals.shift();
      }
      const lines: string[] = [];
      lines.push(`! Generated by Tuning Systems`);
      lines.push(description ?? "");
      lines.push(` ${intervals.length}`);
      lines.push(`!`);
      for (const iv of intervals) {
        lines.push(` ${iv.fraction.toFraction()}`);
      }
      return lines.join("\n") + "\n";
    }

    /**
     * IO-04: clipboard payload — the same data the scaleTable widget renders.
     * TSV (tab-separated) so pasting into spreadsheets gives clean columns.
     */
    export function scalaToCsv(scale: Scale, baseHz: number): string {
      const rows: string[][] = [["Degree", "Ratio", "Cents", "¢ from 12-TET", "Hz"]];
      scale.intervals.forEach((iv, i) => {
        rows.push([
          String(i + 1),
          iv.fraction.toFraction(),
          iv.cents.toFixed(2),
          iv.centsFrom12tet.toFixed(2),
          (baseHz * iv.fraction.valueOf()).toFixed(3),
        ]);
      });
      return rows.map((r) => r.join("\t")).join("\n");
    }

    // -----------------------------------------------------------------------------
    // Internals
    // -----------------------------------------------------------------------------

    function normalizeLines(text: string): string[] {
      // Strip BOM (effective on first line only); normalize CRLF / CR to LF.
      const noBom = text.replace(/^﻿/, "");
      return noBom.replace(/\r\n?/g, "\n").split("\n");
    }

    function stripComment(line: string): string {
      // Comments mid-line are NOT part of the .scl spec (only line-leading `!` is).
      // Trailing whitespace and trailing text after the first token are handled in parsePitchToken.
      return line;
    }

    function parsePitchToken(line: string): Interval {
      // Take the first whitespace-bounded token; ignore everything after.
      const trimmed = line.trim();
      const firstWhitespace = trimmed.search(/\s/);
      const tok = firstWhitespace === -1 ? trimmed : trimmed.slice(0, firstWhitespace);

      if (tok === "") throw new Error(`parsePitchToken: empty token from line "${line}"`);

      // Monzo bra-ket: [a b c> per D-15. Note: D-15 token is NOT whitespace-bounded
      // (components inside [ ... > are space-separated). Re-extract the full bracket form.
      if (trimmed.startsWith("[")) {
        const close = trimmed.indexOf(">");
        if (close === -1) throw new Error(`parsePitchToken: unterminated monzo "${line}"`);
        const inner = trimmed.slice(1, close).trim();
        if (inner === "") return new Interval("1/1");  // empty monzo = 1/1
        const parts = inner.split(/\s+/).map((s) => {
          const n = Number(s);
          if (!Number.isFinite(n) || !Number.isInteger(n)) {
            throw new Error(`parsePitchToken: monzo component "${s}" is not an integer`);
          }
          return n;
        });
        return Interval.fromMonzo(parts);
      }

      // Cents detection (D-19): contains `.` → cents
      if (tok.includes(".")) {
        const cents = Number(tok);
        if (!Number.isFinite(cents)) throw new Error(`parsePitchToken: invalid cents "${tok}"`);
        const ratioFloat = centsToValue(cents);
        // R-01 boundary: centsToValue returns Number; we round-trip via Fraction's defloat
        // approximation. This is LOSSY by definition (cents → ratio cannot recover the original
        // ratio-prime structure). Document accordingly in INVENTORY.
        return new Interval(ratioFloat);
      }

      // Ratio (or bare integer per F06)
      if (tok.startsWith("-")) {
        throw new Error(`parsePitchToken: negative ratio "${tok}" rejected (Huygens-Fokker spec)`);
      }
      if (!/^\d+(\/\d+)?$/.test(tok)) {
        throw new Error(`parsePitchToken: invalid ratio "${tok}" (multi-slash or non-numeric)`);
      }
      // Bare integer → "N/1"
      const ratioStr = tok.includes("/") ? tok : `${tok}/1`;
      return new Interval(ratioStr);
    }
    ```
