    Step 2 — Implement src/audio/synth.ts based on RESEARCH.md lines 469–576 with the disposed-flag adjustment from the test contract above:

    ```ts
    /**
     * Cell-owned synth lifecycle wrapper around sw-synth (ARCHITECTURE Pattern 4).
     *
     * Lazy AudioContext: the AudioContext is NOT created at module load, NOT created at
     * createSynth() call time — it is created on the first playNote / playArpeggio /
     * startDrone call. This satisfies the browser user-gesture requirement and prevents
     * AudioContext leaks under hot-reload (Pitfall #2).
     *
     * Voice tracking: every noteOn returns a noteOff callback (sw-synth contract). We
     * wrap it in a setTimeout-based release for one-shot notes; for drones, we return
     * the noteOff to the caller. activeVoices is a live counter for tests + dev assertions.
     *
     * Dispose is TERMINAL: after dispose(), all method calls are no-ops. The cell pattern
     * (createSynth() inside its own cell, invalidation.then(synth.dispose)) ensures a fresh
     * handle is created on cell re-evaluation.
     */

    import { Synth as SwSynth, defaultParams, type OscillatorVoiceParams } from "sw-synth";

    export interface SynthHandle {
      playNote(hz: number, dur?: number): () => void;
      playNotes(freqs: number[], dur?: number): void;
      playArpeggio(freqs: number[], stepSec?: number): void;
      startDrone(hz: number): () => void;
      panic(): void;
      readonly activeVoices: number;
      dispose(): void;
    }

    export interface CreateSynthOpts {
      master?: number;
      maxPolyphony?: number;
      voiceParams?: Partial<OscillatorVoiceParams>;
    }

    const DEFAULT_NOTE_DUR = 1.5;   // D-18
    const DEFAULT_ARP_STEP = 0.45;  // D-18
    const ARP_NOTE_RATIO = 0.95;    // D-18: noteLen = stepSec * 0.95

    export function createSynth(opts: CreateSynthOpts = {}): SynthHandle {
      let ctx: AudioContext | null = null;
      let master: GainNode | null = null;
      let synth: SwSynth | null = null;
      let activeVoices = 0;
      let disposed = false;

      const ensure = (): boolean => {
        if (disposed) return false;
        if (ctx) return true;
        // Resolve AudioContext constructor (handle webkitAudioContext fallback for older Safari)
        const win = (globalThis as unknown as { window?: { AudioContext?: typeof AudioContext; webkitAudioContext?: typeof AudioContext } }).window;
        const Ctx = win?.AudioContext ?? win?.webkitAudioContext;
        if (!Ctx) {
          // Test environments without AudioContext on globalThis.window: also try direct global
          const direct = (globalThis as unknown as { AudioContext?: typeof AudioContext }).AudioContext;
          if (!direct) return false;
        }
        const Klass = Ctx ?? (globalThis as unknown as { AudioContext: typeof AudioContext }).AudioContext;
        ctx = new Klass({ latencyHint: "interactive" });
        master = ctx.createGain();
        master.gain.value = opts.master ?? 0.2;
        master.connect(ctx.destination);
        synth = new SwSynth(ctx, master);
        synth.maxPolyphony = opts.maxPolyphony ?? 16;
        synth.voiceParams = {
          ...defaultParams(),
          attackTime: 0.005,
          decayTime: 0.030,
          sustainLevel: 0.7,
          releaseTime: 0.150,
          ...opts.voiceParams,
        };
        return true;
      };

      const playNoteImpl = (hz: number, dur: number): (() => void) => {
        if (!ensure() || !synth) return () => {};
        const off = synth.noteOn(hz, 0.7);
        activeVoices++;
        let released = false;
        const release = (): void => {
          if (released) return;
          released = true;
          off();
          activeVoices = Math.max(0, activeVoices - 1);
        };
        const timer = setTimeout(release, dur * 1000);
        return () => {
          clearTimeout(timer);
          release();
        };
      };

      return {
        playNote(hz, dur = DEFAULT_NOTE_DUR) {
          return playNoteImpl(hz, dur);
        },
        playNotes(freqs, dur = DEFAULT_NOTE_DUR) {
          if (!ensure()) return;
          for (const hz of freqs) playNoteImpl(hz, dur);
        },
        playArpeggio(freqs, stepSec = DEFAULT_ARP_STEP) {
          if (!ensure()) return;
          const noteLen = stepSec * ARP_NOTE_RATIO;
          freqs.forEach((hz, i) => {
            if (i === 0) {
              playNoteImpl(hz, noteLen);
            } else {
              setTimeout(() => playNoteImpl(hz, noteLen), i * stepSec * 1000);
            }
          });
        },
        startDrone(hz) {
          if (!ensure() || !synth) return () => {};
          const off = synth.noteOn(hz, 0.5);
          activeVoices++;
          let released = false;
          return () => {
            if (released) return;
            released = true;
            off();
            activeVoices = Math.max(0, activeVoices - 1);
          };
        },
        panic() {
          if (!synth) return;
          synth.allNotesOff();
          activeVoices = 0;
        },
        get activeVoices() {
          return activeVoices;
        },
        dispose() {
          if (disposed) return;
          disposed = true;
          synth?.allNotesOff();
          master?.disconnect();
          ctx?.close().catch(() => {});
          ctx = null;
          master = null;
          synth = null;
          activeVoices = 0;
        },
      };
    }
    ```
