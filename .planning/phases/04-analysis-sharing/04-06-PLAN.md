    `jiSubsetOfEdoCentsBased(N)` — internal helper for the 12tet/19edo/31edo presets. EDO comparisons against JI go through a CENTS-based scale (since 12tet has no rational ratios for most degrees). Implementation:
    ```typescript
    function edoScale(N: number): Scale {
      // N equal divisions of 2/1 — every step at (1200/N)*step cents.
      // Use Interval(centsToValue(c)) — same lossy float→Fraction conversion as
      // parseScala's cents path. The kernel still treats it as a Fraction; for
      // EDO comparisons the cents value is what matters (alignment is by cents,
      // common-subset for EDOs vs JI is naturally rare).
      const intervals: Interval[] = [];
      for (let step = 1; step <= N; step++) {
        const cents = (1200 / N) * step;
        // Round-trip via cents-to-value; matches Phase 2 parseScala's cents handling.
        intervals.push(new Interval(2 ** (cents / 1200)));
      }
      return new Scale(intervals, new Interval("2/1"));
    }
    ```
    Note: the `2 ** (cents / 1200)` is a float; the Interval constructor wraps it via fraction.js, which approximates the float as a rational. For EDO scales this is the standard approach (matches Scale Workshop's representation) and acknowledges the cents-source lossiness per Pitfall #1's rule — display projection is acceptable, the underlying fraction is "best rational approximation" of the float. Document this in the function header comment.

       ```typescript
       interface AlignedRow {
         aDegree: number;
         aInterval: Interval;
         bMatch: Interval | null;        // nullable when B is empty
         centsDelta: number | null;       // |a.cents - bMatch.cents|; null when bMatch is null
         exactMatch: boolean;             // BigInt equality via Interval.equals
       }
       function align(a: Scale, b: Scale): AlignedRow[] {
         return a.intervals.map((aIv, i) => {
           if (b.intervals.length === 0) return { aDegree: i + 1, aInterval: aIv, bMatch: null, centsDelta: null, exactMatch: false };
           let best = b.intervals[0]!;
           let bestDist = Math.abs(aIv.cents - best.cents);
           for (let j = 1; j < b.intervals.length; j++) {
             const cur = b.intervals[j]!;
             const d = Math.abs(aIv.cents - cur.cents);
             if (d < bestDist) { best = cur; bestDist = d; }
           }
           return {
             aDegree: i + 1,
             aInterval: aIv,
             bMatch: best,
             centsDelta: bestDist,
             exactMatch: aIv.equals(best),  // BigInt — D-32, Pitfall #1
           };
         });
       }
       ```
