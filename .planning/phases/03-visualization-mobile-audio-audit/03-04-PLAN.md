**deriveLatticeBasis(scale)** (RESEARCH lines 826-853 verbatim — export so the test stub for octave-only-scale empty-state can use it directly if needed; tests in Plan 01 do call this implicitly via the empty-state assertion):
```ts
export function deriveLatticeBasis(scale: Scale): number[] {
  const used = new Set<number>();
  for (const iv of scale.intervals) {
    iv.monzo.forEach((exp, i) => {
      if (exp !== 0 && i > 0) {
        const p = PRIMES[i];
        if (p !== undefined) used.add(p); // noUncheckedIndexedAccess guard
      }
    });
  }
  const basis = [...used].sort((a, b) => a - b);
  if (basis.length > 3) {
    // Top-2-by-frequency truncation. Recount to rank.
    const counts = new Map<number, number>();
    for (const iv of scale.intervals) {
      iv.monzo.forEach((exp, i) => {
        if (exp !== 0 && i > 0) {
          const p = PRIMES[i];
          if (p !== undefined) counts.set(p, (counts.get(p) ?? 0) + 1);
        }
      });
    }
    const sorted = [...basis].sort((a, b) => (counts.get(b) ?? 0) - (counts.get(a) ?? 0));
    const dropped = sorted.slice(2);
    // Console-warn the dropped primes (per D-19 — "flag remaining via console").
    // eslint-disable-next-line no-console
    console.warn(
      `[lattice] basis truncated from ${sorted.length} to 2 primes; dropped: ${dropped.join(', ')}`,
    );
    return sorted.slice(0, 2);
  }
  return basis;
}
```

**Basis projection helper** (zero-out monzo components NOT in basis — RESEARCH lines 339-348):
```ts
function projectMonzos(monzos: number[][], basis: number[]): number[][] {
  const basisIndices = new Set<number>();
  for (const p of basis) {
    const idx = PRIMES.indexOf(p);
    if (idx >= 0) basisIndices.add(idx);
  }
  return monzos.map((m) => m.map((v, i) => (basisIndices.has(i) ? v : 0)));
}
```

```ts
export function lattice(
  scale: Scale,
  synth: SynthHandle,
  opts: LatticeOpts = {},
): HTMLElement {
  const root = document.createElement("section");
  root.className = "lattice-widget";

  // Heading + helper (UI-SPEC copywriting line 116, 119 — exact strings).
  const heading = document.createElement("h2");
  heading.textContent = "Lattice";
  root.appendChild(heading);

  const helper = document.createElement("p");
  helper.className = "dashboard-helper";
  helper.textContent = "Click a node to audition. Scroll or pinch to zoom; drag to pan.";
  root.appendChild(helper);

  // Pitfall #11 — octave-only scale empty-state (UI-SPEC line 122 — exact copy).
  const basis = opts.basis ?? deriveLatticeBasis(scale);
  if (basis.length === 0) {
    const empty = document.createElement("p");
    empty.className = "lattice-empty";
    empty.textContent = "This scale only spans the octave (prime 2). Add a non-octave interval to see a lattice.";
    root.appendChild(empty);
    return root;
  }

  // Compute coordinates ONCE per render (RESEARCH Pitfall 6 — never inside zoom handler).
  const monzos = scale.intervals.map((iv) => iv.monzo);
  const projected = projectMonzos(monzos, basis);
  const layoutOptions = kraigGrady9();
  // showContext: 'none' -> maxDistance: 0; 'neighbors' -> 1 (default); 'full' -> 2.
  const showContext = opts.showContext ?? 'neighbors';
  layoutOptions.maxDistance = showContext === 'none' ? 0 : showContext === 'full' ? 2 : 1;

  const { vertices, edges } = spanLattice(projected, layoutOptions);

  // Build SVG.
  const width = opts.width ?? DEFAULT_WIDTH;
  const height = opts.height ?? DEFAULT_HEIGHT;
  const svg = d3
    .create("svg")
    .attr("class", "viz lattice")
    .attr("viewBox", `${-width / 2} ${-height / 2} ${width} ${height}`)
    .attr("preserveAspectRatio", "xMidYMid meet")
    .attr("width", width)
    .attr("height", height);

  const g = svg.append("g");

  // Edges first (so nodes render on top).
  g.selectAll("line.edge")
    .data(edges)
    .enter()
    .append("line")
    .attr("class", (d: Edge) => `edge edge-${d.type}`)
    .attr("x1", (d: Edge) => d.x1)
    .attr("y1", (d: Edge) => d.y1)
    .attr("x2", (d: Edge) => d.x2)
    .attr("y2", (d: Edge) => d.y2);

  // Audition handler — bound at handler creation time so 'dyad' vs 'note' is honored.
  // baseHz: caller provides effective baseHz via SynthHandle's owner. lattice does NOT
  // accept a baseHz parameter (D-08 signature); the synth has its own internal Hz
  // path or, where required, the keyboard widget passes baseHz directly.
  // For the lattice's "1/1 + ratio" dyad: we need a Hz reference for 1/1. Since
  // D-08 omits baseHz from lattice's signature, the convention is: pass the
  // effective Hz of 1/1 via a SynthHandle wrapper OR compute via the click
  // handler's caller. The dashboard cell in Plan 06 will create lattice with
  // an audition wrapper around synth that pre-multiplies by effectiveBaseHz.
  //
  // RESOLUTION (consistent with PATTERNS lines 270-282 + audio-panel.ts):
  // Pass an effectiveBaseHz THROUGH the synth interface — by extending LatticeOpts
  // with a `baseHz?: number` for THIS WIDGET'S use. Default 440. The dashboard
  // cell passes its computed effectiveBaseHz here when calling lattice(...).
  // (Adjust the LatticeOpts interface above to include `baseHz?: number;`.)
  //
  // ⇒ Update LatticeOpts to add: `baseHz?: number; // default 440 per Phase 2 D-08`
  // ⇒ Default at usage site: `const baseHz = opts.baseHz ?? 440;`

  const baseHz = opts.baseHz ?? 440;
  const audition = opts.audition ?? 'dyad';

  const auditionVertex = (vertex: Vertex): void => {
    if (vertex.index === undefined) return; // neighbor / out-of-scale: no audition
    const iv = scale.intervals[vertex.index];
    if (!iv) return;
    const ratio = Number(iv.fraction.valueOf()); // S-3 boundary coercion
    if (audition === 'note') {
      synth.playNotes([baseHz * ratio], DEFAULT_AUDITION_DUR_SEC);
    } else {
      synth.playNotes([baseHz, baseHz * ratio], DEFAULT_AUDITION_DUR_SEC);
    }
  };

  // Node groups (per D-21: filled when in-scale, outlined when neighbor).
  const nodeGroups = g
    .selectAll("g.lattice-node")
    .data(vertices)
    .enter()
    .append("g")
    .attr("class", (d: Vertex) =>
      d.index !== undefined ? "lattice-node lattice-node--in-scale" : "lattice-node lattice-node--neighbor",
    )
    .attr("transform", (d: Vertex) => `translate(${d.x},${d.y})`)
    .attr("tabindex", (d: Vertex) => (d.index !== undefined ? 0 : -1))
    .attr("role", (d: Vertex) => (d.index !== undefined ? "button" : "presentation"))
    .attr("aria-label", (d: Vertex) => {
      if (d.index === undefined) return "";
      const iv = scale.intervals[d.index];
      if (!iv) return "";
      return `Play ${iv.fraction.toFraction()}`;
    });

  // Render circles + labels per node.
  nodeGroups.append("circle").attr("r", 18);

  nodeGroups
    .filter((d: Vertex) => d.index !== undefined)
    .each(function (this: SVGGElement, d: Vertex) {
      if (d.index === undefined) return;
      const iv = scale.intervals[d.index];
      if (!iv) return;
      const sel = d3.select(this);
      sel.append("text").attr("class", "ratio").attr("text-anchor", "middle").attr("dy", "0.35em").text(iv.fraction.toFraction());
      const cents = iv.centsFrom12tet;
      const sign = cents >= 0 ? "+" : "−";
      const formatted = `${sign}${Math.abs(cents).toFixed(1)}¢`;
      sel.append("text").attr("class", "cents").attr("text-anchor", "middle").attr("y", 30).text(formatted);
      sel.append("title").text(`${iv.fraction.toFraction()} | ${formatted} from 12-TET`);
    });

  // Click handler.
  nodeGroups.on("click", function (this: SVGGElement, _event: Event, d: Vertex) {
    auditionVertex(d);
  });

  // Keyboard: Enter / Space audition for accessibility (UI-SPEC line 183).
  nodeGroups.on("keydown", function (this: SVGGElement, event: KeyboardEvent, d: Vertex) {
    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      auditionVertex(d);
    }
  });

  // d3.zoom — apply transform to inner <g>, not to <svg>.
  const zoom = d3
    .zoom<SVGSVGElement, unknown>()
    .scaleExtent([0.25, 8])
    .on("zoom", (event: { transform: { toString(): string } }) => {
      g.attr("transform", event.transform.toString());
    });
  svg.call(zoom);

  const svgNode = svg.node();
  if (svgNode) root.appendChild(svgNode);
  return root;
}
```
