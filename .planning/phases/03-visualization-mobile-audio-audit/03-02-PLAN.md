4. For `n` in `[firstKey, lastKey]`:

   - S-3 boundary coercion: `ratio = Number(iv.fraction.valueOf()) * Math.pow(periodRatio, octaveCount)`.

5. Return the Map.

**enumerateDiamond (RESEARCH lines 800-814 verbatim):**
```ts
export function enumerateDiamond(oddLimit: number, scale: Scale): DiamondCell[] {
  // Defense-in-depth bounds (mirror kbm.ts spirit; UI offers {7, 9, 11, 13, 15, 21, 31}).
  if (!Number.isInteger(oddLimit) || oddLimit < 1 || oddLimit > 1023) {
    throw new RangeError(`enumerateDiamond: oddLimit must be integer in [1, 1023] (got ${oddLimit})`);
  }
  const odds: number[] = [];
  for (let k = 1; k <= oddLimit; k += 2) odds.push(k);

  const cells: DiamondCell[] = [];
  for (const i of odds) {
    for (const j of odds) {
      // octaveReduce() default period = 2/1 (Interval.octaveReduce signature).
      // Do NOT pass scale.period — diamond is octave-bound by definition.
      const ratio = new Interval(`${i}/${j}`).octaveReduce();
      const inScale = scale.intervals.some((iv) => iv.equals(ratio));
      cells.push({ numerator: i, denominator: j, ratio, inScale });
    }
  }
  return cells;
}
```
