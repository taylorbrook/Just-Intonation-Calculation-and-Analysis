```ts
const ODD_LIMIT_PRESETS = [7, 9, 11, 13, 15, 21, 31] as const;

export function deriveDiamondOddLimit(scale: Scale): number {
  // For each interval, compute its odd-limit via lib/monzo.ts oddLimit.
  // (Avoid importing oddLimit from xen-dev-utils directly — go through src/lib/monzo.ts.)
  // The scale interval already exposes monzo; primeLimit + oddLimit are both available
  // via PROJECTION onto the existing primitives. Specifically: max odd factor of
  // numerator and denominator after stripping 2s. Use the named-export from monzo.ts.
  // (If oddLimit is not yet re-exported by src/lib/monzo.ts, import directly via
  // monzo.ts; verify before coding — INVENTORY.md row confirms oddLimit is custom.)
  let maxOdd = 1;
  for (const iv of scale.intervals) {
    // oddLimit lives in src/lib/monzo.ts — import alongside primeLimit at the top.
    // (Add to imports at top of this file: import { primeLimit, oddLimit, PRIMES } from "../lib/monzo.js";)
    const ol = oddLimit(iv.monzo);
    if (ol > maxOdd) maxOdd = ol;
  }
  for (const preset of ODD_LIMIT_PRESETS) {
    if (preset >= maxOdd) return preset;
  }
  return ODD_LIMIT_PRESETS[ODD_LIMIT_PRESETS.length - 1] ?? 31;
}
```

Skeleton (steal from lattice.ts and adapt):
```ts
export function tonalityDiamond(
  scale: Scale,
  synth: SynthHandle,
  opts: DiamondOpts = {},
): HTMLElement {
  const root = document.createElement("section");
  root.className = "tonality-diamond-widget";

  const heading = document.createElement("h2");
  heading.textContent = "Tonality diamond";
  root.appendChild(heading);

  const helper = document.createElement("p");
  helper.className = "dashboard-helper";
  helper.textContent = "Click a cell to audition. Hover for ratio details.";
  root.appendChild(helper);

  const limit = opts.oddLimit ?? deriveDiamondOddLimit(scale);
  const cells: DiamondCell[] = enumerateDiamond(limit, scale);

  const showContext = opts.showContext ?? 'neighbors';
  const renderCells = showContext === 'none' ? cells.filter((c) => c.inScale) : cells;

  // Square layout: indexOf(i in odds) → row; indexOf(j in odds) → col.
  const odds: number[] = [];
  for (let k = 1; k <= limit; k += 2) odds.push(k);
  const rankOf = new Map<number, number>();
  odds.forEach((n, i) => rankOf.set(n, i));

  const baseHz = opts.baseHz ?? 440;
  const audition = opts.audition ?? 'dyad';

  const auditionCell = (cell: DiamondCell): void => {
    if (!cell.inScale && showContext !== 'full') return; // out-of-scale not auditioned by default
    const ratio = Number(cell.ratio.fraction.valueOf());
    if (audition === 'note') synth.playNotes([baseHz * ratio], DEFAULT_AUDITION_DUR_SEC);
    else synth.playNotes([baseHz, baseHz * ratio], DEFAULT_AUDITION_DUR_SEC);
  };

  const width = opts.width ?? Math.min(DEFAULT_WIDTH, odds.length * CELL_SIZE + 80);
  const height = opts.height ?? Math.min(DEFAULT_HEIGHT, odds.length * CELL_SIZE + 80);

  const svg = d3
    .create("svg")
    .attr("class", "viz diamond")
    .attr("viewBox", `0 0 ${width} ${height}`)
    .attr("preserveAspectRatio", "xMidYMid meet")
    .attr("width", width)
    .attr("height", height);

  const g = svg.append("g").attr("transform", `translate(${(width - odds.length * CELL_SIZE) / 2},${(height - odds.length * CELL_SIZE) / 2})`);

  // Cells.
  const cellGroups = g
    .selectAll("g.diamond-cell")
    .data(renderCells)
    .enter()
    .append("g")
    .attr("class", (d: DiamondCell) => {
      const pl = primeLimit(d.ratio.monzo);
      const axis = classForPrimeLimit(pl);
      return `diamond-cell ${d.inScale ? "diamond-cell--in-scale" : "diamond-cell--out"} diamond-cell--${axis}`;
    })
    .attr("transform", (d: DiamondCell) => {
      const row = rankOf.get(d.numerator) ?? 0;
      const col = rankOf.get(d.denominator) ?? 0;
      return `translate(${col * CELL_SIZE},${row * CELL_SIZE})`;
    })
    .attr("tabindex", (d: DiamondCell) => (d.inScale || showContext !== 'none' ? 0 : -1))
    .attr("role", (d: DiamondCell) => (d.inScale || showContext !== 'none' ? "button" : "presentation"))
    .attr("aria-label", (d: DiamondCell) =>
      `Play ${d.ratio.fraction.toFraction()}, ${d.inScale ? "in scale" : "not in scale"}`,
    );

  cellGroups.append("rect").attr("width", CELL_SIZE - 2).attr("height", CELL_SIZE - 2).attr("rx", 4);

  cellGroups
    .append("text")
    .attr("class", "ratio")
    .attr("x", (CELL_SIZE - 2) / 2)
    .attr("y", (CELL_SIZE - 2) / 2)
    .attr("text-anchor", "middle")
    .attr("dy", "0.35em")
    .text((d: DiamondCell) => `${d.numerator}/${d.denominator}`);

  // Tooltip via SVG <title> — keyboard-discoverable per D-22.
  cellGroups.append("title").text((d: DiamondCell) => {
    const pl = primeLimit(d.ratio.monzo);
    const sign = d.ratio.cents >= 0 ? "+" : "−";
    const cents = `${sign}${Math.abs(d.ratio.cents).toFixed(1)}¢`;
    return `${d.ratio.fraction.toFraction()} | ${cents} | ${pl}-limit | ${d.inScale ? "in scale" : "not in scale"}`;
  });

  cellGroups.on("click", function (this: SVGGElement, _event: Event, d: DiamondCell) {
    auditionCell(d);
  });
  cellGroups.on("keydown", function (this: SVGGElement, event: KeyboardEvent, d: DiamondCell) {
    if (event.key === "Enter" || event.key === " ") {
      event.preventDefault();
      auditionCell(d);
    }
  });

  const zoom = d3
    .zoom<SVGSVGElement, unknown>()
    .scaleExtent([0.5, 6])
    .on("zoom", (event: { transform: { toString(): string } }) => {
      g.attr("transform", event.transform.toString());
    });
  svg.call(zoom);

  const svgNode = svg.node();
  if (svgNode) root.appendChild(svgNode);
  return root;
}
```

```ts
function formatSignedCents(cents: number, precision: number): string {
  const sign = cents >= 0 ? "+" : "−"; // U+2212 minus sign per UI-SPEC examples ("+3.9¢", "−13.7¢")
  return `${sign}${Math.abs(cents).toFixed(precision)}¢`;
```

**keyboard() factory:**
```ts
export function keyboard(
  scale: Scale,
  synth: SynthHandle,
  baseHz: number,
  opts: KeyboardOpts = {},
): HTMLElement {
  const root = document.createElement("section");
  root.className = "keyboard-widget";

  const heading = document.createElement("h2");
  heading.textContent = "Keyboard";
  root.appendChild(heading);

  const helper = document.createElement("p");
  helper.className = "dashboard-helper";
  helper.textContent = "Click a key to audition. Cents shown are deviation from 12-TET.";
  root.appendChild(helper);

  const precision = opts.precision ?? DEFAULT_PRECISION;
  const keyWidth = opts.keyWidth ?? DEFAULT_KEY_WIDTH;
  const N = scale.intervals.length;
  const totalWidth = keyWidth * N + 8; // + small margin for the period marker
  const totalHeight = DEFAULT_KEY_HEIGHT + 40; // + cents-label band above

  const svgNS = "http://www.w3.org/2000/svg";
  const svg = document.createElementNS(svgNS, "svg");
  svg.setAttribute("class", "viz keyboard");
  svg.setAttribute("viewBox", `0 0 ${totalWidth} ${totalHeight}`);
  svg.setAttribute("preserveAspectRatio", "xMidYMid meet");
  svg.setAttribute("width", String(totalWidth));
  svg.setAttribute("height", String(totalHeight));

  // For each interval, render: cents label above + rectangle key + ratio label inside.
  for (let i = 0; i < N; i++) {
    const iv = scale.intervals[i];
    if (!iv) continue;
    const x = i * keyWidth;
    const centsText = formatSignedCents(iv.centsFrom12tet, precision);
    const ratioText = iv.fraction.toFraction();

    // Key group.
    const keyG = document.createElementNS(svgNS, "g");
    keyG.setAttribute("class", "keyboard__key");
    keyG.setAttribute("transform", `translate(${x},40)`);
    keyG.setAttribute("tabindex", "0");
    keyG.setAttribute("role", "button");
    keyG.setAttribute("aria-pressed", "false");
    keyG.setAttribute(
      "aria-label",
      `Play degree ${i + 1}: ${ratioText}, ${centsText} from 12-TET`,
    );

    const rect = document.createElementNS(svgNS, "rect");
    rect.setAttribute("class", "keyboard__key-rect");
    rect.setAttribute("width", String(keyWidth - 2));
    rect.setAttribute("height", String(DEFAULT_KEY_HEIGHT));
    keyG.appendChild(rect);

    const ratioLabel = document.createElementNS(svgNS, "text");
    ratioLabel.setAttribute("class", "keyboard__ratio");
    ratioLabel.setAttribute("x", String((keyWidth - 2) / 2));
    ratioLabel.setAttribute("y", String(DEFAULT_KEY_HEIGHT - 16));
    ratioLabel.setAttribute("text-anchor", "middle");
    ratioLabel.textContent = ratioText;
    keyG.appendChild(ratioLabel);

    // Cents label ABOVE the key (per UI-SPEC line 198): position at the top band, x-aligned with key center.
    const centsLabel = document.createElementNS(svgNS, "text");
    centsLabel.setAttribute("class", "keyboard__cents");
    centsLabel.setAttribute("x", String(x + (keyWidth - 2) / 2));
    centsLabel.setAttribute("y", "30");
    centsLabel.setAttribute("text-anchor", "middle");
    centsLabel.textContent = centsText;
    svg.appendChild(centsLabel);

    // Pointerdown sustain — release callback is held in a closure-local variable.
    let release: (() => void) | null = null;
    const onDown = (): void => {
      if (release) return; // already held — ignore re-entrant downs
      const ratio = Number(iv.fraction.valueOf()); // S-3 boundary
      release = synth.playNote(baseHz * ratio, KEYBOARD_NOTE_DUR_SEC);
      keyG.setAttribute("aria-pressed", "true");
      keyG.classList.add("keyboard__key--active");
    };
    const onUp = (): void => {
      if (release) {
        release();
        release = null;
      }
      keyG.setAttribute("aria-pressed", "false");
      keyG.classList.remove("keyboard__key--active");
    };
    keyG.addEventListener("pointerdown", onDown);
    keyG.addEventListener("pointerup", onUp);
    keyG.addEventListener("pointerleave", onUp);
    keyG.addEventListener("pointercancel", onUp);

    // Keyboard activation — Enter/Space → fire-and-forget short note (no release semantics for keyboard).
    keyG.addEventListener("keydown", (event) => {
      if (event.key === "Enter" || event.key === " ") {
        event.preventDefault();
        const ratio = Number(iv.fraction.valueOf());
        synth.playNote(baseHz * ratio, KEYBOARD_KEYBOARD_ACTIVATION_DUR_SEC);
      }
    });

    svg.appendChild(keyG);
  }

  // Period boundary marker — vertical dashed line after the last key.
  const marker = document.createElementNS(svgNS, "line");
  marker.setAttribute("class", "keyboard__period-boundary");
  marker.setAttribute("x1", String(N * keyWidth - 1));
  marker.setAttribute("y1", "40");
  marker.setAttribute("x2", String(N * keyWidth - 1));
  marker.setAttribute("y2", String(40 + DEFAULT_KEY_HEIGHT));
  marker.setAttribute("stroke-dasharray", "4 4");
  svg.appendChild(marker);

  root.appendChild(svg);
  return root;
}
```

**TS strict notes:** `scale.intervals[i]` returns `Interval | undefined` — guard with `if (!iv) continue` before each per-key render. Use `document.createElementNS` for SVG-namespaced elements (NOT `document.createElement` — the latter creates HTML-namespace nodes that browsers may render but vitest/happy-dom treats them as non-SVG; `createElementNS` is explicit and correct).
