export class Scale {
  readonly intervals: readonly Interval[]; // intervals[0] = 1/1; LAST = period (D-14)
  readonly period: Interval;
  rotate(degree: number): Scale;  // degree 0 = identity copy; rotate(n) = n-th mode, starts on 1/1, sorted

}

- `const SVG_NS = "http://www.w3.org/2000/svg";` then `document.createElementNS(SVG_NS, "svg"|"g"|"circle"|"line"|"text"|"title")`.
- Every numeric setAttribute wrapped in `String(...)` (strict TS).
- `if (!s) continue;` / `if (!iv) continue;` guards for `noUncheckedIndexedAccess`.
- Labels via `textContent` only — NEVER innerHTML.
- Signed cents use U+2212 (true minus "−"), not U+002D hyphen.
- Click-audition: `g.addEventListener("click", () => synth.playNote(baseHz * ratio, 0.6))`; Enter/Space keydown mirror with `e.preventDefault()`.
